cmake_minimum_required(VERSION 3.15)
project(GoldPickup)

# Force static linking of the MSVC C/C++ Runtime Library (/MT)
# This ensures users don't need the Visual C++ Redistributable installed
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")

# Define the shared library (DLL) and its source files
# (Adjust the paths if your files are in a /src/ folder)
add_library(GoldPickup SHARED 
    src/DLLmain.cpp
    src/GoldPickup.cpp
)

# Set standard Windows definitions
target_compile_definitions(GoldPickup PRIVATE WIN32 _WINDOWS _CRT_SECURE_NO_DEPRECATE)
